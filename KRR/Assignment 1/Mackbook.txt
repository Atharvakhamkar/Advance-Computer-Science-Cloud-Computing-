%% Propositions: MacBookInKitchen, SmartSpeakerFlashing, IHearMusic, ListeningToRadio,

%% Use only the specified vocabulary, otherwise the autograder will not work.

%% A1: "If MacBook is offline and I hear music, then I must be listening to the radio."

       -(MacBookOnline) & IHearMusic -> ListeningToRadio.

%% A2: "The MacBook is always in hotspot range of my Motorola except when one is in the
%%      kitchen and the other in the bathroom."

        MacBookInHotSpotRangeMotorola <-> -((MacBookInKitchen & MotorolaInBathroom) | (MacBookInBathroom & MotorolaInKitchen)).

%% A3: "The MacBook is online if and only if either: (a) the WiFi is up; or (b) the
%%      Motorola is connected to 5G and the Motorola is in hotspot range."

        MacBookOnline <-> (WiFiUp | (MotorolaOn5G & MacBookInHotSpotRangeMotorola)).

%% A4: "If the smart speaker light is flashing, the WiFi is down."

        SmartSpeakerFlashing -> -WiFiUp.

%% A5: "If the the Motorola is in the kitchen, it is either on the window sill or on the
%%      comfy chair."

        MotorolaInKitchen -> (MotorolaOnWindowSill | MotorolaOnComfyChair).

%% A6: "The Motorola is neither on the window sill nor on the comfy chair."

        -MotorolaOnWindowSill & -MotorolaOnComfyChair.

%% A7: "The Motorola is in one of three rooms: kitchen or lounge, or bathroom."

        MotorolaInKitchen | MotorolaInLounge | MotorolaInBathroom.

%% A8: "The MacBook is in the kitchen, the smart speaker light is flashing and I hear
%%      music."

        MacBookInKitchen & SmartSpeakerFlashing & IHearMusic.



%% GOAL: "The Motorola is in the lounge, unless I'm listening to the radio."

          -ListeningToRadio -> MotorolaInLounge.
